cmake_minimum_required(VERSION 3.24)
project(windows-dns-config-auditor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(WSA_ANALYZE "MSVC static analysis" OFF)
option(WSA_ASAN "Address sanitizer" OFF)
add_library(audit_core src/model.cpp src/address.cpp src/analysis.cpp src/format.cpp src/cli.cpp)
target_include_directories(audit_core PUBLIC src)
target_compile_features(audit_core PUBLIC cxx_std_17)
target_compile_definitions(audit_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(audit_windows src/acquire.cpp src/adapters.cpp src/dns_servers.cpp src/suffix.cpp src/registration.cpp)
target_link_libraries(audit_windows PUBLIC audit_core iphlpapi advapi32)
add_executable(windows-dns-config-auditor src/main.cpp src/app.manifest)
target_link_libraries(windows-dns-config-auditor PRIVATE audit_windows)
foreach(t audit_core audit_windows windows-dns-config-auditor)
 target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
 if(WSA_ANALYZE)
  target_compile_options(${t} PRIVATE /analyze)
 endif()
 if(WSA_ASAN)
  target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_executable(audit_tests tests/tests.cpp)
 target_link_libraries(audit_tests PRIVATE audit_core)
 target_compile_options(audit_tests PRIVATE /W4 /WX /utf-8)
 if(WSA_ASAN)
  target_compile_options(audit_tests PRIVATE /fsanitize=address /Zi)
 endif()
 add_executable(nonadmin_tests tests/nonadmin.cpp)
 target_link_libraries(nonadmin_tests PRIVATE audit_windows)
 target_compile_options(nonadmin_tests PRIVATE /W4 /WX /utf-8)
 if(WSA_ASAN)
  target_compile_options(nonadmin_tests PRIVATE /fsanitize=address /Zi)
 endif()
 add_executable(nonadmin_launch tests/nonadmin_launch.cpp)
 target_compile_features(nonadmin_launch PRIVATE cxx_std_17)
 target_compile_definitions(nonadmin_launch PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_options(nonadmin_launch PRIVATE /W4 /WX /utf-8 /EHsc)
 target_link_libraries(nonadmin_launch PRIVATE advapi32)
 add_test(NAME synthetic COMMAND audit_tests)
 add_test(NAME live_json COMMAND windows-dns-config-auditor --json)
 add_test(NAME help COMMAND windows-dns-config-auditor --help)
 add_test(NAME version COMMAND windows-dns-config-auditor --version)
endif()
install(TARGETS windows-dns-config-auditor RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/windows-dns-config-auditor)


install(FILES VALIDATION.md schemas/schema-v1.0.json DESTINATION share/windows-dns-config-auditor)
